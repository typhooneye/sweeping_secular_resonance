def cal_ssr_curve(a_pert, e_pert, 
                  m_pert, dens0, kg,   
                apert_in, apert_out, da=0.01,
                  force_apert_in=None):
   
    #apert_in, apert_out = cal_gap(a_pert, m_pert, e_pert)
    #if force_apert_in is None:
    #    apert_in = 5 * r_star

    a_ssr, t_ssr, sigma = [], [], []
    a_vals = np.arange(a_pert*0.1, a_pert*2 + da, da)

    for a_p in a_vals:
        g_jd = Gpd(m_pert, a_pert, apert_in, apert_out, kg)
        g_pd = Gpd(m_emb, a_p, apert_in, apert_out, kg)
        g_pj = Gij(a_p, a_pert, m_pert)
        g_pg = Ggr(a_p)

        denom = g_jd - g_pd
        if denom == 0:
            continue

        expt = (g_pj + g_pg) / denom
        tssr = -np.log(expt)
        if tssr > 0:
            a_ssr.append(a_p)
            t_ssr.append(tssr)
            sigma.append(dens0 * expt)

    return (np.array(a_ssr), np.array(t_ssr), np.array(sigma),
            apert_in, apert_out)
