def integrand(x, s, n, alpha):
  r = math.cos(n*x) / (1 + alpha**2 - 2*alpha*math.cos(x))**s
  r = r/math.pi
  return r

def bsn(s, n, alpha):
  b = quad(integrand, 0 , 2*math.pi, args = (s, n, alpha))
  bf = b[0]
  return bf

def integrand1(r, dens):
  result = 2 * math.pi * r * dens * r**(-kg)
  return result

def cal_mj(dens, kg, r1, r2):
  mj = quad(integrand1, r1, r2, args = (dens, kg))
  return mj[0]

def Gij(ai, aj, mj):
  alpha = min(ai, aj) / max(ai, aj)
  b1 = bsn(1.5, 1, alpha)
  ni = ai**(-1.5) * m_star**0.5
  gij = mj * alpha * b1 
  gij = gij / (4 * ni * ai**2 * max(ai, aj))
  return gij

def cal_gap(aj, mj, ej):
  ratio = mj/m_star
  rh = (ratio/3)**0.333
  if ej > rh:
    return aj*(1-ej)*(1-rh), aj*(1+ej)*(1+rh)
  else:
    return aj*(1-rh), aj*(1+rh)
